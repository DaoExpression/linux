===============================================================================
       FREEBSD VNET JAIL & INTERNAL HOMELAB NETWORK CONFIGURATION GUIDE
            (Hardened Internet-Facing PF & Multi-Port SSH Redirection)
===============================================================================

This guide configures an isolated internal virtual network on a FreeBSD host
exposed securely to the public internet using a hardened Packet Filter (PF)
firewall. It includes rate-limiting tables, anti-spoofing protections, and 
distinct external ports redirected to specific internal jails.

Architecture Overview:
----------------------
  Physical Interface (WAN):    em0 (e.g., public IP: 203.0.113.50)
  Internal Bridge (Switch):    bridge0 (Gateway: 10.0.0.1/24)
  Internal Subnet:             10.0.0.0/24

  Jails & Mapped Ports:
    1. Host Management SSH:    External Port 2200 -> Host (127.0.0.1:22)
    2. "web" (FreeBSD Nginx)   -> External Port 2222 -> 10.0.0.10:22
    3. "db"  (FreeBSD Database)-> External Port 2223 -> 10.0.0.20:22
    4. "linux" (Debian/Ubuntu) -> External Port 2224 -> 10.0.0.30:22

===============================================================================
SECTION 1: HOST SYSTEM PREPARATION (/etc/rc.conf & /etc/sysctl.conf)
===============================================================================

1. Enable necessary services and virtual network modules in /etc/rc.conf:

-------------------------------------------------------------------------------
# /etc/rc.conf snippet
-------------------------------------------------------------------------------
jail_enable="YES"
pf_enable="YES"
pf_rules="/etc/pf.conf"
pflog_enable="YES"
cloned_interfaces="bridge0"
ifconfig_bridge0="inet 10.0.0.1 netmask 255.255.255.0 description 'Jail-Internal-Bridge'"
linux_enable="YES"


2. Enable IP forwarding and kernel hardening in /etc/sysctl.conf:

-------------------------------------------------------------------------------
# /etc/sysctl.conf snippet
-------------------------------------------------------------------------------
# Enable IPv4 Packet Forwarding (Host acts as gateway for Jails)
net.inet.ip.forwarding=1

# Allow raw sockets in jails if needed (ping, traceroute)
security.jail.allow_raw_sockets=1

# Kernel Security & ASLR
kern.elf64.aslr.enable=1
kern.elf32.aslr.enable=1
security.bsd.see_other_uids=0
security.bsd.see_other_gids=0


===============================================================================
SECTION 2: HARDENED FIREWALL & NAT SETUP WITH SSH REDIRECTION (/etc/pf.conf)
===============================================================================

This production-grade pf.conf uses tables for brute-force mitigation, normalizes
packets, provides NAT for outbound jail internet access, and safely forwards
distinct external ports to individual internal jails.

-------------------------------------------------------------------------------
# /etc/pf.conf
-------------------------------------------------------------------------------
# 1. Macros & Interfaces
ext_if   = "em0"                  # Change to your public network interface
jail_if  = "bridge0"
jail_net = "10.0.0.0/24"

# Internal Jail IPs
host_ip  = "127.0.0.1"
web_jail = "10.0.0.10"
db_jail  = "10.0.0.20"
linux_jail = "10.0.0.30"

# 2. Tables for Brute-Force Protection
# IPs that hammer SSH ports get automatically added here and blocked
table <bad_hosts> persist

# 3. Options & Normalization
set skip on lo0
set block-policy drop
set loginterface $ext_if

scrub in all reassemble tcp

# 4. Anti-Spoofing Protections
antispoof quick for { $ext_if $jail_if }

# 5. NAT (Network Address Translation) for Internal Jails Outbound Traffic
nat on $ext_if from $jail_net to any -> ($ext_if)

# 6. Port Forwarding / Redirection Rules (Inbound)
# Redirect distinct external ports to respective internal jail SSH ports with inline pass rules
rdr pass on $ext_if proto tcp from any to ($ext_if) port 2200 -> $host_ip    port 22
rdr pass on $ext_if proto tcp from any to ($ext_if) port 2222 -> $web_jail   port 22
rdr pass on $ext_if proto tcp from any to ($ext_if) port 2223 -> $db_jail    port 22
rdr pass on $ext_if proto tcp from any to ($ext_if) port 2224 -> $linux_jail port 22

# Optional: Port Forward HTTP/HTTPS traffic to the Web Jail
rdr pass on $ext_if proto tcp from any to ($ext_if) port 80  -> $web_jail port 80
rdr pass on $ext_if proto tcp from any to ($ext_if) port 443 -> $web_jail port 443

# 7. Global Filter Rules
block all

# Drop traffic immediately from known bad hosts table
block quick from <bad_hosts>

# Pass all internal bridge traffic freely between jails and host gateway
pass quick on $jail_if all

# Pass outgoing traffic from host and NAT gateway
pass out quick on $ext_if all keep state

# Pass incoming connections to forwarded SSH ports with brute-force rate limiting protection
# If an IP tries to open more than 3 new SSH connections per 60 seconds, flag and drop them into <bad_hosts>
pass in quick on $ext_if proto tcp to any port { 2200, 2222, 2223, 2224 } \
    flags S/SA keep state \
    (max-src-conn-rate 3/60, overload <bad_hosts> flush global)


===============================================================================
SECTION 3: JAIL CONFIGURATION MASTER FILE (/etc/jail.conf)
===============================================================================

-------------------------------------------------------------------------------
# /etc/jail.conf
-------------------------------------------------------------------------------
path = "/usr/local/jails/$name";
host.hostname = "$name.home.internal";

exec.start = "/bin/sh /etc/rc";
exec.stop = "/bin/sh /etc/rc.shutdown";
exec.clean;
mount.devfs;

vnet;
vnet.interface = "epair${vnet_id}b";

exec.prestart  = "jt=`mktemp -t jail` && devfs ruleset -s 4 load user > $jt && devfs -m /usr/local/jails/$name/dev ruleset 4 apply && rm $jt";
exec.created   = "ifconfig epair${vnet_id} create up";
exec.created  += "ifconfig bridge0 addm epair${vnet_id}a up";
exec.poststop  = "ifconfig epair${vnet_id}a destroy";
devfs_ruleset = "4";

# 1. FreeBSD Web Jail
web {
    $vnet_id = "10";
    exec.start += "/sbin/ifconfig epair10b inet 10.0.0.10 netmask 255.255.255.0 up";
    exec.start += "/sbin/route add default 10.0.0.1";
}

# 2. FreeBSD Database Jail
db {
    $vnet_id = "20";
    exec.start += "/sbin/ifconfig epair20b inet 10.0.0.20 netmask 255.255.255.0 up";
    exec.start += "/sbin/route add default 10.0.0.1";
}

# 3. Linux Jail (Debian/Ubuntu)
linux {
    $vnet_id = "30";
    host.hostname = "linux.home.internal";
    allow.raw_sockets;
    sysvmsg=new;
    syssem=new;
    sysvshm=new;
    mount.fstab = "/etc/jail.fstab.linux";
    exec.start = "/bin/true";
    exec.stop = "/bin/true";
    exec.poststart += "jexec linux ifconfig epair30b 10.0.0.30 netmask 255.255.255.0 up";
    exec.poststart += "jexec linux route add default gw 10.0.0.1";
}

===============================================================================
SECTION 5: TESTING & MANAGEMENT COMMANDS
===============================================================================

1. Test PF configuration syntax before applying:
   $ sudo pfctl -nf /etc/pf.conf

2. Load and reload firewall rules live:
   $ sudo pfctl -f /etc/pf.conf

3. Check blocked brute-force IPs in the table:
   $ sudo pfctl -t bad_hosts -s show

4. Manually clear a locked-out IP from the bad hosts table:
   $ sudo pfctl -t bad_hosts -d 203.0.113.99

5. Connect externally to specific jailed instances:
   $ ssh -p 2222 root@your_public_ip   # Connects to Web Jail (10.0.0.10)
   $ ssh -p 2223 root@your_public_ip   # Connects to DB Jail (10.0.0.20)
   $ ssh -p 2224 root@your_public_ip   # Connects to Linux Jail (10.0.0.30)
